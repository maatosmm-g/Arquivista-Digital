Resumo Executivo (Clique para Editar)
Este código representa a base de uma plataforma de organização de notebooks digitais ("Gaveta Digital"). O sistema utiliza Firebase para persistência de dados em tempo real, com uma arquitetura robusta de segurança baseada em regras de acesso (Firestore Rules). As principais funcionalidades incluem: organização automática de categorias via IA (Gemini), compartilhamento público de itens individuais, galeria pública por usuário e telemetria de acessos para links compartilhados.

export const notebookConverter: FirestoreDataConverter<Notebook> = {
  toFirestore: (notebook: Notebook) => ({
    userId: notebook.userId,
    name: notebook.name,
    category: notebook.category,

    content: notebook.content || '',
    createdAt: notebook.createdAt,
    updatedAt: notebook.updatedAt || null,

      category: data.category,

      content: data.content,
      createdAt: data.createdAt?.toDate?.() || data.createdAt,
      updatedAt: data.updatedAt?.toDate?.() || data.updatedAt,
      lastAnalyzed: data.lastAnalyzed?.toDate?.() || data.lastAnalyzed,
    };
  },
};

export class FirestoreError extends Error {
  public readonly operationType: OperationType;
  public readonly path: string | null;
  public readonly authInfo: AuthInfo;

  constructor(
    message: string,
    operationType: OperationType,
    path: string | null,
    authInfo: AuthInfo
  ) {
    super(message);
    this.name = 'FirestoreError';
    this.operationType = operationType;
    this.path = path;
    this.authInfo = authInfo;
    if (Error.captureStackTrace) {
      Error.captureStackTrace(this, FirestoreError);
    }
  }

  toJSON() {
    return {
      name: this.name,
      message: this.message,
      operationType: this.operationType,
      path: this.path,
      authInfo: this.authInfo,
      stack: this.stack,
    };
  }
}

export enum OperationType {
  CREATE = 'create',
  UPDATE = 'update',
  DELETE = 'delete',
  LIST = 'list',
  GET = 'get',
  WRITE = 'write',
}

export interface AuthInfo {
  userId?: string | null;
  email?: string | null;
  emailVerified?: boolean | null;
  isAnonymous?: boolean | null;
  tenantId?: string | null;
  providerInfo?: { providerId?: string | null; email?: string | null; }[];
}

export async function testConnection(): Promise<boolean> {
  try {
    await getDocFromServer(doc(db, 'test', 'connection'));
    console.log('✅ Firebase connection successful');
    return true;
  } catch (error) {
    if (error instanceof Error && error.message.includes('the client is offline')) {
      console.error('❌ Firebase connection failed: Client is offline');
    } else if (error instanceof Error && error.message.includes('permission')) {
      console.warn('⚠️ Firebase connected but lacks permissions for test collection');
      return true;
    }
    console.error('❌ Firebase connection error:', error);
    return false;
  }
}

export function handleFirestoreError(
  error: unknown,
  operationType: OperationType,
  path: string | null
): never {
  const authInfo: AuthInfo = {
    userId: auth.currentUser?.uid,
    email: auth.currentUser?.email,
    emailVerified: auth.currentUser?.emailVerified,
    isAnonymous: auth.currentUser?.isAnonymous,
    tenantId: auth.currentUser?.tenantId,
    providerInfo: auth.currentUser?.providerData?.map(provider => ({
      providerId: provider.providerId,
      email: provider.email,
    })) || []
  };

  const errorMessage = error instanceof Error ? error.message : String(error);
  const firestoreError = new FirestoreError(errorMessage, operationType, path, authInfo);
  console.error('Firestore Error:', firestoreError.toJSON());
  throw firestoreError;
}

import { useState, useEffect } from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';

import { onAuthStateChanged, User, signInWithPopup } from 'firebase/auth';
import { collection, query, where, onSnapshot, orderBy, doc, setDoc, serverTimestamp } from 'firebase/firestore';
import { Notebook, notebookConverter } from './lib/types';
import { NotebookCard } from './components/NotebookCard';
import { PublicGallery } from './components/PublicGallery';
import { Plus, Wand2, Loader2, Share2 } from 'lucide-react';

function Dashboard() {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);
  const [notebooks, setNotebooks] = useState<Notebook[]>([]);
  const [isReorganizing, setIsReorganizing] = useState(false);
  const [showShareTooltip, setShowShareTooltip] = useState(false);

  // Teste de conexão explícito
  useEffect(() => {
    testConnection();
  }, []);

  const copyGalleryLink = () => {
    if (!user) return;
    const url = `${window.location.origin}/gallery/${user.uid}`;
    navigator.clipboard.writeText(url);
    setShowShareTooltip(true);
    setTimeout(() => setShowShareTooltip(false), 3000);
  };

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, async (u) => {
      setUser(u);
      setLoading(false);
      if (u) {
        const userRef = doc(db, 'users', u.uid);
        await setDoc(userRef, {
          uid: u.uid, email: u.email, displayName: u.displayName,
          photoURL: u.photoURL, lastLoginAt: serverTimestamp(), createdAt: serverTimestamp(),
        }, { merge: true }).catch(console.error);
      }
    });
    return unsubscribe;
  }, []);

  useEffect(() => {
    if (!user) {
      setNotebooks([]);
      return;
    }
    const q = query(
      collection(db, 'notebooks'),
      where('userId', '==', user.uid),
      orderBy('createdAt', 'desc')
    ).withConverter(notebookConverter); // ✅ Conversor elimina any[]

    return onSnapshot(q, (snapshot) => {
      setNotebooks(snapshot.docs.map(doc => doc.data()));
    });
  }, [user]);

  if (loading) return (<div className="flex items-center justify-center min-h-screen"><Loader2 className="w-8 h-8 animate-spin" /></div>);
  if (!user) return (
    <div className="flex flex-col items-center justify-center min-h-screen gap-4">
      <h1 className="text-2xl font-serif">Gaveta Digital</h1>
      <button onClick={() => signInWithPopup(auth, googleProvider)} className="px-6 py-3 bg-black text-white rounded-lg">Entrar com Google</button>
    </div>
  );

  return (
    <div className="max-w-7xl mx-auto px-4 py-8">
      <div className="flex justify-between items-center mb-8">
        <div><h1 className="text-3xl font-serif">Minha Gaveta</h1><p className="text-gray-600">{notebooks.length} notebooks</p></div>
        <div className="flex gap-3">
          <button onClick={copyGalleryLink} className="flex items-center gap-2 px-4 py-2 border rounded-lg"><Share2 className="w-4 h-4" /> Compartilhar Galeria</button>

🃏 src/components/NotebookCard.tsx Cards com tipagem completa e logs 
import React, { useState } from 'react';
import { Notebook, AccessLog } from '../lib/types';
import { db } from '../lib/firebase';
import { doc, updateDoc, deleteDoc, collection, query, where, getDocs } from 'firebase/firestore';
import { Globe, Lock, Trash2, Eye, ChevronDown, ChevronUp } from 'lucide-react';

interface NotebookCardProps {
  notebook: Notebook;
  isSharedView?: boolean;
}

export const NotebookCard: React.FC<NotebookCardProps> = ({ notebook, isSharedView = false }) => {
  const [isPublic, setIsPublic] = useState(notebook.isPublic);
  const [showLogs, setShowLogs] = useState(false);
  const [logs, setLogs] = useState<AccessLog[]>([]);

  const fetchLogs = async () => {
    if (!isSharedView && showLogs === false) {
      const q = query(collection(db, 'access_logs'), where('notebookId', '==', notebook.id));
      const snapshot = await getDocs(q);
      setLogs(snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() } as AccessLog)));
    }
    setShowLogs(!showLogs);
  };

  const togglePublic = async () => {
    if (isSharedView) return;
    await updateDoc(doc(db, 'notebooks', notebook.id), { isPublic: !isPublic });
    setIsPublic(!isPublic);
  };

  const handleDelete = async () => {
    if (confirm(`Excluir "${notebook.name}"?`)) {
      await deleteDoc(doc(db, 'notebooks', notebook.id));
    }
  };

  return (
    <div className="bg-white border rounded-lg p-4 shadow-sm">
      <div className="flex justify-between items-start">
        <h3 className="font-semibold text-lg">{notebook.name}</h3>
        {!isSharedView && (
          <div className="flex gap-2">
            <button onClick={togglePublic}>{isPublic ? <Globe className="w-4 h-4" /> : <Lock className="w-4 h-4" />}</button>
            <button onClick={handleDelete} className="text-red-600"><Trash2 className="w-4 h-4" /></button>
          </div>
        )}
      </div>
      <span className="inline-block px-2 py-1 bg-gray-100 rounded-full text-xs mt-2">{notebook.category}</span>
      {notebook.content && <p className="text-gray-600 text-sm mt-2">{notebook.content.substring(0, 100)}...</p>}
      
      {!isSharedView && (
        <button onClick={fetchLogs} className="flex items-center gap-1 text-xs text-gray-500 mt-3">
          <Eye className="w-3 h-3" /> {showLogs ? <ChevronUp className="w-3 h-3" /> : <ChevronDown className="w-3 h-3" />} Logs de acesso
        </button>
      )}
      {showLogs && !isSharedView && (
        <div className="mt-2 text-xs text-gray-500">
          {logs.length === 0 ? 'Nenhum acesso registrado' : logs.map(log => <div key={log.id}>{log.action} em {log.timestamp?.toString()}</div>)}
        </div>
      )}
    </div>
  );
};
🛡️ firestore.rules Regras de segurança completas 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // REGRAS PARA NOTEBOOKS
    match /notebooks/{notebookId} {
      allow read: if request.auth != null && (
        resource.data.userId == request.auth.uid ||
        resource.data.isPublic == true
      );
      
      allow create: if request.auth != null &&
        request.resource.data.keys().hasAll(['userId', 'name', 'category']) &&
        request.resource.data.userId == request.auth.uid;
      
      allow update: if request.auth != null &&
        resource.data.userId == request.auth.uid;
      
      allow delete: if request.auth != null &&
        resource.data.userId == request.auth.uid;
    }
    
    // REGRAS PARA USUÁRIOS
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow create, update: if request.auth != null && request.auth.uid == userId;
      allow delete: if false;
    }
    
    // REGRAS PARA LOGS DE ACESSO
    match /access_logs/{logId} {
      allow create: if request.auth != null;
      
      allow read: if request.auth != null && 
        exists(/databases/$(database)/documents/notebooks/$(request.resource.data.notebookId)) &&
        get(/databases/$(database)/documents/notebooks/$(request.resource.data.notebookId)).data.userId == request.auth.uid;
      
      allow update, delete: if false;
    }
    
    // BLOQUEIO GLOBAL DE SEGURANÇA
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
